internal class MyList
{
//Class này chưa danh sách các sinh viên. Có thể lấy từ Database lên, hoặc hard-code
List<Student> _ds = new List<Student>()
{
new Student(){Id = "SE1", Address = "Dĩ An", Name = "An", Gpa = 8.8, Yob = 2003},
new Student(){Id = "SE2", Address = "Bình Dương", Name = "Bình", Gpa = 9.0, Yob = 2008},

new Student(){Id = "SE4", Address = "Châu Thành", Name = "Dũng", Gpa = 5.0, Yob = 2006},
new Student(){Id = "SE4", Address = "Long An", Name = "Thành", Gpa = 8.0, Yob = 2000},
}; //danh sách sv khởi tạo sẵn luôn
//Lấy ra danh sách sinh viên có GPA >= 8
//Trong code phải có chỗ so sánh điểm của từng sinh viên
//coi lớn hơn bằng 8 hay không
public List<Student> GetStudentsByGpa()
{
List<Student> result = new(); //Viết full: new List<Student>
//quét qua từng bạn
foreach (Student item in _ds)
{
if (item.Gpa >= 8) //condition, điều kiện kiểm soát data
//predicate, hàm, lệnh dự đoán
{
		            result.Add(item);
}
}
return result;
}
//Lấy ra danh sách sinh viên có GPA >= 8 and Yob >= 2005
public List<Student> GetStudentsByGpaAndYob()
{
List<Student> result = new(); //Viết full: new List<Student>
//quét qua từng bạn
foreach (Student item in _ds)
{
if (item.Gpa >= 8 && item.Yob >= 2005) //condition, điều kiện kiểm soát data
//predicate, hàm, lệnh dự đoán
{
		            result.Add(item);
}
}
return result;
}
}

static void getStudentsByAnyCondition()
{

        Console.WriteLine(student.ToString());
